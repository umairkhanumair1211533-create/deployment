// Task Manager Application // All functionality is handled via JavaScript with event listeners // ============================================= // DOM ELEMENT REFERENCES // ============================================= const taskInput = document.getElementById('taskInput'); const addButton = document.getElementById('addButton'); const taskList = document.getElementById('taskList'); const taskCount = document.getElementById('taskCount'); // ============================================= // LOCAL STORAGE FUNCTIONS // ============================================= /** * Saves the current list of tasks to localStorage * Converts the task list items to an array of task objects * Each task object contains: { text: string, completed: boolean } */ function saveTasksToStorage() { const tasks = []; const listItems = taskList.querySelectorAll('li'); listItems.forEach(li => { const taskText = li.querySelector('.task-text').textContent; const isCompleted = li.classList.contains('completed'); tasks.push({ text: taskText, completed: isCompleted }); }); localStorage.setItem('tasks', JSON.stringify(tasks)); } /** * Loads tasks from localStorage and renders them in the task list * If no tasks exist in storage, does nothing * Called automatically when the page loads */ function loadTasksFromStorage() { const storedTasks = localStorage.getItem('tasks'); if (!storedTasks) return; try { const tasks = JSON.parse(storedTasks); tasks.forEach(task => { createTaskElement(task.text, task.completed); }); updateTaskCount(); } catch (error) { console.error('Error loading tasks from localStorage:', error); } } // ============================================= // TASK CREATION AND MANAGEMENT FUNCTIONS // ============================================= /** * Creates a new task list item (li) with the given text * @param {string} taskText - The text content of the task * @param {boolean} isCompleted - Whether the task should be marked as completed * @returns {HTMLElement} The newly created li element * * Each task li contains: * - A span with class 'task-text' for the task text (clicking toggles completed) * - A delete button with text 'x' (clicking removes the task) */ function createTaskElement(taskText, isCompleted = false) { // Create the list item const li = document.createElement('li'); if (isCompleted) { li.classList.add('completed'); } // Create the task text span const taskSpan = document.createElement('span'); taskSpan.className = 'task-text'; taskSpan.textContent = taskText; // Create the delete button const deleteBtn = document.createElement('button'); deleteBtn.className = 'delete-btn'; deleteBtn.textContent = 'x'; // Add event listener to toggle completed status when task text is clicked taskSpan.addEventListener('click', function() { toggleTaskCompletion(li); }); // Add event listener to delete the task when delete button is clicked deleteBtn.addEventListener('click', function() { deleteTask(li); }); // Assemble the list item li.appendChild(taskSpan); li.appendChild(deleteBtn); // Add the task to the task list taskList.appendChild(li); return li; } /** * Adds a new task to the task list * Validates that the input is not empty before adding * Clears the input field after successful addition * Updates the task count and saves to localStorage */ function addTask() { const taskText = taskInput.value.trim(); // Don't add empty tasks if (taskText === '') { return; } // Create the task element createTaskElement(taskText, false); // Clear the input field taskInput.value = ''; // Update the task count updateTaskCount(); // Save tasks to localStorage saveTasksToStorage(); } /** * Toggles the 'completed' class on a task list item * Updates the task count and saves to localStorage after toggling * @param {HTMLElement} li - The list item element to toggle */ function toggleTaskCompletion(li) { li.classList.toggle('completed'); updateTaskCount(); saveTasksToStorage(); } /** * Deletes a task from the task list * Updates the task count and saves to localStorage after deletion * @param {HTMLElement} li - The list item element to remove */ function deleteTask(li) { li.remove(); updateTaskCount(); saveTasksToStorage(); } /** * Updates the task count paragraph to show how many tasks are not completed * Counts all task items and subtracts completed tasks * Displays the count in the format: "Tasks remaining: X" */ function updateTaskCount() { const allTasks = taskList.querySelectorAll('li'); const completedTasks = taskList.querySelectorAll('li.completed'); const remainingTasks = allTasks.length - completedTasks.length; taskCount.textContent = `Tasks remaining: ${remainingTasks}`; } // ============================================= // EVENT LISTENERS // ============================================= // Add task when the Add button is clicked addButton.addEventListener('click', addTask); // Add task when Enter key is pressed in the input field taskInput.addEventListener('keypress', function(event) { if (event.key === 'Enter') { addTask(); } }); // ============================================= // INITIALIZATION // ============================================= // Load tasks from localStorage when the page loads loadTasksFromStorage();
